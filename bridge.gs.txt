/*
 * 우리 달력 -> 개인 Google Calendar 단방향 자동 동기화
 *
 * 설치 경로
 * 1. 우리 달력에서 window.UriBridge.open({dbUrl, code, me})를 열고 본인을 직접 선택합니다.
 * 2. '내 설정이 포함된 스크립트 복사'를 누릅니다.
 * 3. https://script.google.com 에서 새 프로젝트를 만들고 Code.gs에 붙여넣습니다.
 * 4. 프로젝트 설정의 시간대를 Asia/Seoul로 지정합니다.
 * 5. 왼쪽 '서비스 +'에서 Calendar API를 추가합니다. 기본 Cloud 프로젝트의 API도 함께 활성화됩니다.
 * 6. 함수 목록에서 setupUriBridge를 선택해 한 번 실행하고 Google 권한을 승인합니다.
 *
 * 동작
 * - 공유 방의 events만 읽고, 새로 만든 전용 캘린더에 한 방향으로 추가/수정/삭제합니다.
 * - 기본 캘린더나 다른 캘린더의 일정은 읽지 않습니다.
 * - 15분 트리거, 실행 잠금, 결정적 Google 일정 ID로 중복과 동시 실행을 막습니다.
 * - 한국 시간 기준 오늘 일정이 있는 날에는 전용 캘린더에 짧은 알림 일정을 만들고
 *   5분 전 Google Calendar 팝업을 붙입니다. 종일 일정도 포함합니다.
 * - 관리자 키, Firebase 비밀키, 웹 앱 배포, 이메일 권한은 사용하지 않습니다.
 *
 * 중지: stopUriBridge() 실행. 전용 캘린더와 기존 복사 일정은 남습니다.
 * 재개: setupUriBridge() 실행.
 * 완전 제거: removeUriBridgeData(true) 실행. 전용 캘린더도 삭제됩니다.
 */

const URI_BRIDGE_USER = Object.freeze({
  dbUrl: '__URI_DB_URL__',
  code: '__URI_ROOM_CODE__',
  me: '__URI_ME__',
  notificationTime: '09:05'
});

const URI_BRIDGE = Object.freeze({
  version: '1.0.0',
  timezone: 'Asia/Seoul',
  calendarName: '우리 달력 (자동 동기화)',
  configKey: 'URI_BRIDGE_CONFIG',
  statusKey: 'URI_BRIDGE_STATUS',
  syncFunction: 'syncUriBridge',
  maxEvents: 3000,
  noticeDaysAhead: 45
});

/** 처음 연결 또는 재개할 때 사용자가 Apps Script 편집기에서 직접 실행합니다. */
function setupUriBridge() {
  const user = validateUserConfig_();
  const props = PropertiesService.getScriptProperties();
  const old = readProperty_(URI_BRIDGE.configKey, {});
  let calendar = old.calendarId ? CalendarApp.getCalendarById(old.calendarId) : null;
  if (!calendar) {
    calendar = CalendarApp.createCalendar(URI_BRIDGE.calendarName, {
      description: '우리 달력 공유 방 events의 개인용 단방향 복사본입니다. 기본 캘린더는 읽지 않습니다.',
      timeZone: URI_BRIDGE.timezone
    });
  }
  const config = {
    version: URI_BRIDGE.version,
    dbUrl: user.dbUrl,
    code: user.code,
    me: user.me,
    notificationTime: user.notificationTime,
    calendarId: calendar.getId(),
    enabled: true,
    configuredAt: new Date().toISOString()
  };
  props.setProperty(URI_BRIDGE.configKey, JSON.stringify(config));
  replaceSyncTrigger_();
  const result = syncUriBridge();
  console.log('설정 완료: 전용 캘린더와 15분 자동 동기화를 시작했습니다.');
  console.log(JSON.stringify(publicStatus_(config, result), null, 2));
  return publicStatus_(config, result);
}

/** 15분마다 실행됩니다. 수동으로 실행해도 됩니다. */
function syncUriBridge() {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return { ok: true, skipped: true, reason: '다른 동기화가 실행 중입니다.' };
  try {
    const config = requireConfig_();
    if (!config.enabled) return { ok: true, skipped: true, reason: '자동 동기화가 중지되어 있습니다.' };

    const source = fetchRoomEvents_(config);
    const desired = buildDesiredEvents_(source, config);
    const existing = listBridgeEvents_(config.calendarId);
    const stats = { added: 0, updated: 0, deleted: 0, unchanged: 0, ignored: source.ignored };

    Object.keys(desired).forEach(function (eventId) {
      const wanted = desired[eventId];
      const current = existing[eventId];
      if (!current) {
        calendarRequest_('post', config.calendarId, '', wanted.resource, [200, 201]);
        stats.added++;
      } else if (current.fingerprint !== wanted.fingerprint) {
        calendarRequest_('put', config.calendarId, eventId, wanted.resource, [200]);
        stats.updated++;
      } else {
        stats.unchanged++;
      }
      delete existing[eventId];
    });

    Object.keys(existing).forEach(function (eventId) {
      const old = existing[eventId];
      // 원본에 ID는 남아 있지만 값이 손상된 경우, 일시적 데이터 오류일 수 있으므로 삭제하지 않습니다.
      if (old.kind === 'source' && source.ignoredIds[old.sourceId]) {
        stats.ignored++;
        return;
      }
      calendarRequest_('delete', config.calendarId, eventId, null, [200, 204, 404, 410]);
      stats.deleted++;
    });

    const result = { ok: true, at: new Date().toISOString(), stats: stats };
    PropertiesService.getScriptProperties().setProperty(URI_BRIDGE.statusKey, JSON.stringify(result));
    return result;
  } catch (err) {
    const result = { ok: false, at: new Date().toISOString(), error: errorText_(err) };
    PropertiesService.getScriptProperties().setProperty(URI_BRIDGE.statusKey, JSON.stringify(result));
    console.error(result.error);
    return result;
  } finally {
    lock.releaseLock();
  }
}

function stopUriBridge() {
  deleteSyncTriggers_();
  const props = PropertiesService.getScriptProperties();
  const config = readProperty_(URI_BRIDGE.configKey, null);
  if (config) {
    config.enabled = false;
    config.stoppedAt = new Date().toISOString();
    props.setProperty(URI_BRIDGE.configKey, JSON.stringify(config));
  }
  console.log('자동 동기화를 중지했습니다. 전용 캘린더와 기존 일정은 남아 있습니다.');
  return { ok: true };
}

function getUriBridgeStatus() {
  const config = readProperty_(URI_BRIDGE.configKey, null);
  return publicStatus_(config, readProperty_(URI_BRIDGE.statusKey, null));
}

function removeUriBridgeData(confirmDelete) {
  if (confirmDelete !== true) throw new Error('전용 캘린더까지 지우려면 removeUriBridgeData(true)를 실행하세요.');
  deleteSyncTriggers_();
  const config = readProperty_(URI_BRIDGE.configKey, null);
  if (config && config.calendarId) {
    const calendar = CalendarApp.getCalendarById(config.calendarId);
    if (calendar) calendar.deleteCalendar();
  }
  PropertiesService.getScriptProperties().deleteAllProperties();
  console.log('브리지 설정과 전용 캘린더를 삭제했습니다.');
  return { ok: true };
}

function buildDesiredEvents_(source, config) {
  const desired = {};
  Object.keys(source.events).sort().forEach(function (sourceId) {
    const event = source.events[sourceId];
    const eventId = deterministicId_('event:' + sourceId);
    const resource = sourceResource_(event, config.me, sourceId);
    const fingerprint = fingerprintResource_(resource);
    resource.extendedProperties.private.fingerprint = fingerprint;
    desired[eventId] = { resource: resource, fingerprint: fingerprint };
  });

  const today = Utilities.formatDate(new Date(), URI_BRIDGE.timezone, 'yyyy-MM-dd');
  const last = addDateText_(today, URI_BRIDGE.noticeDaysAhead);
  const byDay = {};
  Object.keys(source.events).forEach(function (id) {
    const event = source.events[id];
    const start = event.start < today ? today : event.start;
    const end = event.end > last ? last : event.end;
    if (start > end) return;
    for (let day = start; day <= end; day = addDateText_(day, 1)) {
      (byDay[day] || (byDay[day] = [])).push(event);
    }
  });

  Object.keys(byDay).forEach(function (day) {
    const resource = noticeResource_(day, byDay[day], config.notificationTime);
    const eventId = deterministicId_('notice:' + day);
    const fingerprint = fingerprintResource_(resource);
    resource.extendedProperties.private.fingerprint = fingerprint;
    desired[eventId] = { resource: resource, fingerprint: fingerprint };
  });
  return desired;
}

function sourceResource_(event, me, sourceId) {
  const owner = event.owner === 'both' ? '함께' : event.owner.toUpperCase();
  const description = [
    event.memo,
    '',
    '담당: ' + owner,
    '내 선택: ' + me.toUpperCase(),
    '우리 달력에서 자동 동기화됨'
  ].filter(function (line, index) { return index > 0 || !!line; }).join('\n');
  const resource = {
    summary: event.title,
    description: description,
    reminders: { useDefault: false },
    extendedProperties: { private: { uriBridge: '1', kind: 'source', sourceId: sourceId, fingerprint: '' } }
  };
  if (event.allDay) {
    resource.start = { date: event.start };
    resource.end = { date: addDateText_(event.end, 1) };
  } else {
    let start = event.start + 'T' + event.st + ':00+09:00';
    let end = event.end + 'T' + (event.et || event.st) + ':00+09:00';
    if (Date.parse(end) <= Date.parse(start)) end = new Date(Date.parse(start) + 60000).toISOString();
    resource.start = { dateTime: start, timeZone: URI_BRIDGE.timezone };
    resource.end = { dateTime: end, timeZone: URI_BRIDGE.timezone };
  }
  resource.id = deterministicId_('event:' + sourceId);
  return resource;
}

function noticeResource_(day, events, notificationTime) {
  events.sort(compareEvents_);
  const parts = notificationTime.split(':').map(Number);
  const startMinutes = parts[0] * 60 + parts[1];
  const endMinutes = startMinutes + 5;
  const endDay = endMinutes >= 1440 ? addDateText_(day, 1) : day;
  const endHm = pad_(Math.floor((endMinutes % 1440) / 60)) + ':' + pad_(endMinutes % 60);
  const lines = events.map(function (event) {
    const time = event.allDay ? '종일' : event.st + (event.et ? '–' + event.et : '');
    return '• [' + time + '] ' + event.title;
  });
  const id = deterministicId_('notice:' + day);
  return {
    id: id,
    summary: '오늘 일정 ' + events.length + '개',
    description: lines.join('\n') + '\n\n우리 달력 당일 알림용 자동 일정입니다.',
    start: { dateTime: day + 'T' + notificationTime + ':00+09:00', timeZone: URI_BRIDGE.timezone },
    end: { dateTime: endDay + 'T' + endHm + ':00+09:00', timeZone: URI_BRIDGE.timezone },
    transparency: 'transparent',
    visibility: 'private',
    reminders: { useDefault: false, overrides: [{ method: 'popup', minutes: 5 }] },
    extendedProperties: { private: { uriBridge: '1', kind: 'notice', sourceId: day, fingerprint: '' } }
  };
}

function listBridgeEvents_(calendarId) {
  const found = {};
  let pageToken = '';
  do {
    const query = '?singleEvents=true&showDeleted=false&maxResults=2500&privateExtendedProperty=' + encodeURIComponent('uriBridge=1') +
      (pageToken ? '&pageToken=' + encodeURIComponent(pageToken) : '');
    const data = calendarRequest_('get', calendarId, query, null, [200]);
    (data.items || []).forEach(function (item) {
      const privateData = item.extendedProperties && item.extendedProperties.private || {};
      found[item.id] = {
        fingerprint: privateData.fingerprint || '',
        kind: privateData.kind || '',
        sourceId: privateData.sourceId || ''
      };
    });
    pageToken = data.nextPageToken || '';
  } while (pageToken);
  return found;
}

function calendarRequest_(method, calendarId, eventIdOrQuery, body, allowed) {
  const isQuery = String(eventIdOrQuery || '').charAt(0) === '?';
  const suffix = isQuery ? eventIdOrQuery : eventIdOrQuery ? '/' + encodeURIComponent(eventIdOrQuery) : '';
  const url = 'https://www.googleapis.com/calendar/v3/calendars/' + encodeURIComponent(calendarId) + '/events' + suffix;
  const options = {
    method: method,
    muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken(), Accept: 'application/json' }
  };
  if (body != null) {
    options.contentType = 'application/json';
    options.payload = JSON.stringify(body);
  }
  const response = UrlFetchApp.fetch(url, options);
  const status = response.getResponseCode();
  if (allowed.indexOf(status) < 0) {
    const hint = status === 403 ? ' Apps Script 왼쪽 서비스 +에서 Calendar API를 추가했는지 확인하세요.' : '';
    throw new Error('Google Calendar 반영 실패 (HTTP ' + status + ').' + hint);
  }
  const text = response.getContentText();
  return text ? JSON.parse(text) : {};
}

function fetchRoomEvents_(config) {
  const url = config.dbUrl + '/couples/' + encodeURIComponent(config.code) + '/events.json';
  const response = UrlFetchApp.fetch(url, { method: 'get', muteHttpExceptions: true, headers: { Accept: 'application/json' } });
  const status = response.getResponseCode();
  if (status !== 200) throw new Error('공유 방 events 읽기 실패 (HTTP ' + status + '). Firebase 규칙과 주소를 확인하세요.');
  let raw;
  try { raw = JSON.parse(response.getContentText() || 'null'); }
  catch (_) { throw new Error('공유 방 응답이 올바른 JSON이 아닙니다.'); }
  if (raw == null) return { events: {}, ignored: 0, ignoredIds: {} };
  if (!plainObject_(raw)) throw new Error('events 데이터 모양이 올바르지 않습니다.');
  if (Object.keys(raw).length > URI_BRIDGE.maxEvents) throw new Error('일정 수가 안전 한도(' + URI_BRIDGE.maxEvents + '개)를 넘었습니다.');
  const events = {};
  const ignoredIds = {};
  let ignored = 0;
  Object.keys(raw).forEach(function (id) {
    const event = cleanEvent_(id, raw[id]);
    if (event) events[id] = event;
    else { ignored++; ignoredIds[String(id)] = true; }
  });
  return { events: events, ignored: ignored, ignoredIds: ignoredIds };
}

function cleanEvent_(id, raw) {
  if (!/^[A-Za-z0-9_-]{1,64}$/.test(String(id || '')) || !plainObject_(raw) || !validDate_(raw.start)) return null;
  const start = raw.start;
  const end = validDate_(raw.end) && raw.end >= start ? raw.end : start;
  const st = validTime_(raw.st) ? raw.st : '';
  const allDay = raw.allDay !== false || !st;
  let et = validTime_(raw.et) ? raw.et : st;
  if (!allDay && end === start && et < st) et = st;
  return {
    id: String(id), title: text_(raw.title, 60) || '(제목 없음)',
    owner: ['a', 'b', 'both'].indexOf(raw.owner) >= 0 ? raw.owner : 'a',
    start: start, end: end, allDay: allDay, st: allDay ? '' : st, et: allDay ? '' : et,
    memo: text_(raw.memo, 500)
  };
}

function validateUserConfig_() {
  const dbUrl = String(URI_BRIDGE_USER.dbUrl || '').trim().replace(/\/+$/, '');
  const code = String(URI_BRIDGE_USER.code || '').trim();
  const me = URI_BRIDGE_USER.me;
  const notificationTime = String(URI_BRIDGE_USER.notificationTime || '09:05');
  if (!/^https:\/\/[a-z0-9-]+(?:\.[a-z0-9-]+)*\.(?:firebaseio\.com|firebasedatabase\.app)$/i.test(dbUrl)) throw new Error('URI_BRIDGE_USER.dbUrl을 확인하세요.');
  if (!/^[a-z0-9]{12,40}$/.test(code)) throw new Error('URI_BRIDGE_USER.code를 확인하세요.');
  if (me !== 'a' && me !== 'b') throw new Error('URI_BRIDGE_USER.me를 a 또는 b로 직접 선택하세요.');
  if (!validTime_(notificationTime)) throw new Error('notificationTime은 HH:MM 형식이어야 합니다.');
  return { dbUrl: dbUrl, code: code, me: me, notificationTime: notificationTime };
}

function replaceSyncTrigger_() {
  deleteSyncTriggers_();
  ScriptApp.newTrigger(URI_BRIDGE.syncFunction).timeBased().everyMinutes(15).create();
}

function deleteSyncTriggers_() {
  ScriptApp.getProjectTriggers().forEach(function (trigger) {
    if (trigger.getHandlerFunction() === URI_BRIDGE.syncFunction) ScriptApp.deleteTrigger(trigger);
  });
}

function requireConfig_() {
  const config = readProperty_(URI_BRIDGE.configKey, null);
  if (!config) throw new Error('setupUriBridge를 먼저 실행하세요.');
  return config;
}

function publicStatus_(config, last) {
  return {
    ok: true,
    configured: !!config,
    enabled: !!(config && config.enabled),
    timezone: URI_BRIDGE.timezone,
    calendarId: config ? config.calendarId : null,
    notificationTime: config ? config.notificationTime : null,
    lastSync: last || null
  };
}

function deterministicId_(value) {
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, value, Utilities.Charset.UTF_8);
  return 'uri' + digest.map(function (b) { return ('0' + ((b + 256) % 256).toString(16)).slice(-2); }).join('').slice(0, 40);
}

function fingerprintResource_(resource) {
  const copy = JSON.parse(JSON.stringify(resource));
  if (copy.extendedProperties && copy.extendedProperties.private) copy.extendedProperties.private.fingerprint = '';
  return deterministicId_(JSON.stringify(copy));
}

function readProperty_(key, fallback) {
  try {
    const value = PropertiesService.getScriptProperties().getProperty(key);
    return value ? JSON.parse(value) : fallback;
  } catch (_) { return fallback; }
}

function compareEvents_(a, b) { return (a.allDay === b.allDay ? 0 : a.allDay ? -1 : 1) || a.st.localeCompare(b.st) || a.title.localeCompare(b.title, 'ko'); }
function addDateText_(ymd, days) { const p = ymd.split('-').map(Number); const d = new Date(Date.UTC(p[0], p[1] - 1, p[2] + days)); return Utilities.formatDate(d, 'UTC', 'yyyy-MM-dd'); }
function pad_(value) { return String(value).padStart(2, '0'); }
function validTime_(value) { return /^(?:[01]\d|2[0-3]):[0-5]\d$/.test(String(value || '')); }
function text_(value, max) { return typeof value === 'string' ? value.trim().slice(0, max) : ''; }
function plainObject_(value) { return !!value && typeof value === 'object' && !Array.isArray(value); }
function errorText_(err) { return err && err.message ? String(err.message) : String(err); }
function validDate_(value) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(value || ''))) return false;
  const p = String(value).split('-').map(Number);
  const d = new Date(Date.UTC(p[0], p[1] - 1, p[2]));
  return p[0] >= 1900 && p[0] <= 2200 && d.getUTCFullYear() === p[0] && d.getUTCMonth() === p[1] - 1 && d.getUTCDate() === p[2];
}
